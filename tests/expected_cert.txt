== tamper the claim (no re-sign): signature fails ==
(tampered (refused bad-signature))

== signed by an untrusted issuer: refused before any re-run ==
(refused untrusted-issuer)

== NEGATIVE CONTROL: valid signature, trusted issuer, WRONG claim ==
   (the re-run refuses what the signature would have waved through)
(wrong-but-signed (refused counterexamples (((-3) "false") ((-1) "false"))))
(impure-but-signed (refused impure ("print: performs I/O")))
